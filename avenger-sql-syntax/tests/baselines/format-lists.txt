FROM samples.penguins

     x => x(flipper_length_mm, scale => scale(zero => FALSE)),
     y => body_mass_g,

   );
FROM t
|> AGGREGATE
     avg(body_mass_g) AS mu,
     stddev(body_mass_g) AS sigma,
     count(*) AS n,
     min(body_mass_g) AS low,
   GROUP BY species, island;
FROM t
|> AGGREGATE count(*) AS n, avg(body_mass_g) AS mean_body_mass_in_grams
   GROUP BY species, island, sex;
FROM t
|> SELECT
     a,
     b,
|> EXTEND [1, 2, 3] AS xs, STRUCT(1 AS a, 'x' AS b) AS s;
FROM t
|> AGGREGATE count(DISTINCT x) AS n, array_agg(x IGNORE NULLS ORDER BY y LIMIT 3) AS xs;
FROM t
|> SELECT - -x, -(-x), + -x, a - -b, -1 AS n
