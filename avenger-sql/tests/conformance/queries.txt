# Pipe queries of the planner's conformance check, each after a line `## <label>`. Each runs after
# tables.sql, which types the CSV tables penguins, sales, and events. GoogleSQL's execute_query
# takes America/Los_Angeles as its default time zone and avenger UTC, so the queries name UTC where
# a timestamp meets a civil time.

## where
FROM penguins |> WHERE island = 'Dream' AND body_mass_g > 4000 |> AGGREGATE COUNT(*) AS n
## where-null
FROM penguins |> WHERE sex IS NULL |> SELECT species, island, body_mass_g
## select-expressions
FROM sales |> SELECT region, units * price AS revenue, units IS NULL AS missing
## extend-set-drop-rename
FROM sales |> EXTEND units * 2 AS doubled |> SET price = price * 10 |> DROP month |> RENAME doubled AS twice |> WHERE region = 'east'
## as-range-variable
FROM sales |> AS s |> WHERE s.units > 3 |> SELECT s.region, s.units
## aggregate-group-by
FROM penguins |> AGGREGATE COUNT(*) AS n, AVG(body_mass_g) AS mean, MIN(bill_length_mm) AS shortest GROUP BY species, island
## aggregate-group-and-order-by
FROM sales |> AGGREGATE SUM(units) AS units GROUP AND ORDER BY region DESC
## aggregate-without-groups
FROM sales |> AGGREGATE COUNT(*) AS n_rows, COUNT(units) AS counted, SUM(units) AS total, COUNT(DISTINCT region) AS regions
## aggregate-rollup
FROM sales |> AGGREGATE SUM(units) AS units GROUP BY ROLLUP(region, product)
## aggregate-grouping-sets
FROM sales |> AGGREGATE COUNT(*) AS n GROUP BY GROUPING SETS ((region), (product), ())
## aggregate-cube
FROM sales |> AGGREGATE COUNT(*) AS n GROUP BY CUBE(region, month)
## aggregate-functions
FROM penguins |> WHERE body_mass_g IS NOT NULL |> AGGREGATE STRING_AGG(DISTINCT island, ',' ORDER BY island) AS islands, ARRAY_AGG(DISTINCT species ORDER BY species) AS kinds, COUNTIF(sex = 'MALE') AS males, LOGICAL_AND(body_mass_g > 2000) AS all_heavy, ANY_VALUE(species IS NOT NULL) AS anything
## aggregate-having-max
FROM sales |> AGGREGATE ANY_VALUE(product HAVING MAX units) AS best GROUP BY region
## order-by-limit
FROM penguins |> WHERE body_mass_g IS NOT NULL |> ORDER BY body_mass_g DESC, bill_length_mm |> LIMIT 5 OFFSET 2 |> SELECT species, body_mass_g
## distinct
FROM sales |> SELECT region |> DISTINCT
## join-using
FROM sales |> JOIN (FROM sales |> AGGREGATE SUM(units) AS total GROUP BY region) USING (region) |> SELECT region, product, units, total
## join-left
FROM (FROM sales |> SELECT DISTINCT region) AS r |> LEFT JOIN (FROM sales |> WHERE product = 'plum') AS p ON r.region = p.region |> SELECT r.region, p.units
## join-full
FROM (FROM sales |> WHERE region = 'east') AS a |> FULL JOIN (FROM sales |> WHERE region = 'west') AS b USING (product, month) |> SELECT product, month, a.units AS east, b.units AS west
## join-cross-comma
FROM (FROM sales |> SELECT DISTINCT region), (FROM sales |> SELECT DISTINCT product) |> AGGREGATE COUNT(*) AS pairs
## union-all
FROM sales |> SELECT region |> UNION ALL (FROM events |> SELECT kind)
## union-distinct-by-name
FROM sales |> SELECT product, region |> UNION DISTINCT BY NAME (FROM sales |> SELECT region, product)
## intersect-except
FROM sales |> SELECT product |> INTERSECT DISTINCT (FROM sales |> WHERE region = 'west' |> SELECT product) |> EXCEPT DISTINCT (FROM sales |> WHERE region = 'north' |> SELECT product)
## intersect-all
FROM sales |> SELECT region |> INTERSECT ALL (FROM sales |> WHERE units > 2 |> SELECT region)
## corresponding
FROM sales |> SELECT region, units |> UNION ALL CORRESPONDING (FROM sales |> SELECT units, product AS region)
## pivot
FROM sales |> SELECT region, product, units |> PIVOT (SUM(units) AS u FOR product IN ('apple', 'pear', 'plum'))
## unpivot
FROM (FROM sales |> AGGREGATE SUM(units) AS units, SUM(price) AS price GROUP BY region) |> UNPIVOT (amount FOR measure IN (units, price))
## unpivot-include-nulls
FROM sales |> WHERE region = 'east' |> SELECT product, month, units |> UNPIVOT INCLUDE NULLS (value FOR k IN (units))
## pipe-with
FROM sales |> WITH totals AS (FROM sales |> AGGREGATE SUM(units) AS total GROUP BY region) |> JOIN totals USING (region) |> SELECT DISTINCT region, total
## scalar-subquery
FROM sales |> SELECT region, units, (FROM sales |> AGGREGATE MAX(units)) AS most
## correlated-exists
FROM (FROM sales |> SELECT DISTINCT region) AS r |> WHERE EXISTS (FROM sales AS s |> WHERE s.region = r.region AND s.product = 'apple')
## in-subquery
FROM penguins |> WHERE species IN (FROM penguins |> WHERE body_mass_g > 6000 |> SELECT species) |> AGGREGATE COUNT(*) AS n
## array-subquery
FROM (FROM sales |> SELECT DISTINCT region) AS r |> SELECT region, ARRAY(FROM sales |> WHERE sales.region = r.region |> SELECT product |> ORDER BY product) AS products
## unnest-cross-join
FROM events |> CROSS JOIN UNNEST(tags) AS tag WITH OFFSET AS pos |> SELECT id, tag, pos
## unnest-from
FROM UNNEST([3, 1, 2]) AS x WITH OFFSET AS i |> SELECT x, i
## unnest-structs
FROM UNNEST([STRUCT('a' AS k, 1 AS v), STRUCT('b', 2)]) |> SELECT k, v * 10 AS w
## window-rank
FROM penguins |> WHERE body_mass_g IS NOT NULL |> EXTEND RANK() OVER (PARTITION BY species ORDER BY body_mass_g DESC) AS r |> WHERE r <= 2 |> SELECT species, body_mass_g, r
## window-running-sum
FROM sales |> EXTEND SUM(units) OVER (PARTITION BY region ORDER BY month ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS running |> SELECT region, month, running
## window-named
FROM sales |> EXTEND AVG(price) OVER w AS mean, COUNT(*) OVER w AS n WINDOW w AS (PARTITION BY product) |> SELECT DISTINCT product, mean, n
## window-lag-lead
FROM sales |> WHERE region = 'east' |> EXTEND LAG(units) OVER (ORDER BY month, product) AS before, LEAD(units, 1, -1) OVER (ORDER BY month, product) AS after |> SELECT product, month, before, after
## window-ntile-percent
FROM penguins |> WHERE body_mass_g IS NOT NULL AND species = 'Gentoo' |> EXTEND NTILE(4) OVER (ORDER BY body_mass_g, bill_length_mm) AS q |> AGGREGATE COUNT(*) AS n GROUP BY q
## case-and-conditionals
FROM sales |> SELECT CASE WHEN units > 4 THEN 'many' WHEN units > 1 THEN 'some' ELSE 'few' END AS amount, IF(region = 'east', 1, 0) AS east, COALESCE(units, 0) AS units, IFNULL(units, -1) AS u, NULLIF(product, 'pear') AS p
## case-value
FROM sales |> SELECT CASE region WHEN 'east' THEN 'E' WHEN 'west' THEN 'W' END AS r
## casts
FROM sales |> SELECT CAST(units AS STRING) AS s, CAST(price AS INT64) AS rounded, CAST(month AS STRING) AS m, SAFE_CAST(product AS INT64) AS bad
## math
FROM sales |> SELECT units / 2 AS half, DIV(units, 2) AS whole, MOD(units, 2) AS odd, ROUND(price * 1.07, 2) AS taxed, ABS(units - 4) AS distance, POW(2, units) AS power, SQRT(units) AS root, GREATEST(units, 3) AS at_least, LEAST(price, 2) AS at_most
## math-rounding
FROM UNNEST([-2.5, -1.5, -0.5, 0.5, 1.5, 2.5]) AS x |> SELECT x, ROUND(x) AS r, TRUNC(x) AS t, CEIL(x) AS c, FLOOR(x) AS f, SIGN(x) AS s
## strings
FROM sales |> SELECT UPPER(region) AS up, LENGTH(product) AS len, CONCAT(region, '-', product) AS key, SUBSTR(product, 2, 2) AS mid, REPLACE(product, 'p', 'P') AS swapped, STARTS_WITH(product, 'p') AS p, STRPOS(product, 'e') AS e
## strings-more
FROM UNNEST(['  Hello World ', 'abc', '']) AS s |> SELECT TRIM(s) AS t, LTRIM(s) AS l, LOWER(s) AS lo, SPLIT(TRIM(s), ' ') AS words
## regexp
FROM UNNEST(['a1b22', 'ccc', 'x9']) AS s |> SELECT REGEXP_CONTAINS(s, r'\d') AS digits, REGEXP_EXTRACT(s, r'(\d+)') AS number, REGEXP_REPLACE(s, r'\d', '#') AS masked
## like
FROM sales |> WHERE product LIKE 'p%' OR region LIKE '%th' |> SELECT region, product
## dates
FROM sales |> SELECT month, EXTRACT(MONTH FROM month) AS m, DATE_ADD(month, INTERVAL 1 MONTH) AS next, DATE_DIFF(DATE '2024-12-31', month, DAY) AS days, DATE_TRUNC(month, QUARTER) AS quarter, FORMAT_DATE('%Y/%m', month) AS label
## timestamps
FROM events |> SELECT id, EXTRACT(HOUR FROM ts AT TIME ZONE 'UTC') AS hour, UNIX_SECONDS(TIMESTAMP_TRUNC(ts, DAY, 'UTC')) AS day, TIMESTAMP_DIFF(ts, TIMESTAMP '2024-03-10 00:00:00+00', MINUTE) AS minutes, DATE(ts, 'UTC') AS d, UNIX_SECONDS(ts) AS seconds
## timestamps-format
FROM events |> SELECT id, FORMAT_TIMESTAMP('%Y-%m-%d %H:%M', ts, 'UTC') AS text, UNIX_SECONDS(TIMESTAMP_ADD(ts, INTERVAL 90 MINUTE)) AS later
## datetime-parse
FROM UNNEST(['2024-02-29', '2023-12-31']) AS s |> SELECT PARSE_DATE('%Y-%m-%d', s) AS d
## arrays
FROM events |> SELECT id, ARRAY_LENGTH(tags) AS n, tags[SAFE_OFFSET(0)] AS first, 'a' IN UNNEST(tags) AS has_a, ARRAY_CONCAT(tags, ['z']) AS more
## array-literals
FROM UNNEST([1, 2, 3]) AS x |> SELECT x, GENERATE_ARRAY(1, x) AS upto, [x, x * x] AS pair, ARRAY_REVERSE([x, 0]) AS back
## structs
FROM sales |> SELECT STRUCT(region AS r, units AS u) AS s |> SELECT s.r, s.u
## struct-field-access
FROM UNNEST([STRUCT(1 AS a, STRUCT('x' AS b) AS nested)]) AS row |> SELECT row.a, row.nested.b
## between-in
FROM sales |> WHERE units BETWEEN 2 AND 5 AND region IN ('east', 'west') AND product NOT IN ('pear') |> SELECT region, product, units
## is-distinct-from
FROM sales |> WHERE units IS DISTINCT FROM 3 |> AGGREGATE COUNT(*) AS n
## logic-with-nulls
FROM UNNEST([TRUE, FALSE, NULL]) AS a CROSS JOIN UNNEST([TRUE, FALSE, NULL]) AS b |> SELECT a, b, a AND b AS both_true, a OR b AS either, NOT a AS na
## tablesample-percent
FROM penguins |> TABLESAMPLE BERNOULLI (100 PERCENT) |> AGGREGATE COUNT(*) AS n
## variables
DECLARE heavy FLOAT64 DEFAULT 5000;
SET heavy = heavy + 500;
FROM penguins |> WHERE body_mass_g >= heavy |> AGGREGATE COUNT(*) AS n
## temp-function
CREATE TEMP FUNCTION revenue(u INT64, p FLOAT64) AS (u * p); FROM sales |> SELECT region, revenue(units, price) AS r
## temp-table
CREATE TEMP TABLE big AS (FROM sales |> WHERE units > 3); FROM big |> AGGREGATE COUNT(*) AS n, SUM(units) AS s
## temp-table-function
CREATE TEMP TABLE FUNCTION top(k INT64) AS (FROM sales |> ORDER BY units DESC |> LIMIT k); FROM top(2) |> SELECT product, units
## parameters-of-literals
FROM UNNEST([1]) |> SELECT 5 / 2 AS q, 7 / 7 AS one, 1e3 AS big, 0x1F AS hex, NUMERIC '1.10' AS n, b'ab' AS bytes, 'it\'s' AS quote, """triple""" AS t, r'\n' AS raw
## initcap
FROM UNNEST(['hello world']) AS s |> SELECT INITCAP(s) AS cap
## repeat
FROM UNNEST(['ab']) AS s |> SELECT REPEAT(s, 2) AS twice
## left-right
FROM UNNEST(['abcdef']) AS s |> SELECT LEFT(s, 3) AS first, RIGHT(s, 3) AS last
## lpad
FROM sales |> SELECT LPAD(product, 6, '*') AS padded
## reverse
FROM UNNEST(['abc', '']) AS s |> SELECT REVERSE(s) AS r
## last-day
FROM UNNEST([DATE '2024-02-10']) AS d |> SELECT LAST_DAY(d) AS last
## array-to-string
FROM events |> SELECT id, ARRAY_TO_STRING(tags, '+') AS joined
## grouping
FROM sales |> AGGREGATE SUM(units) AS u, GROUPING(region) AS gr, GROUPING(product) AS gp GROUP BY ROLLUP(region, product)
## error-sqrt
FROM UNNEST([4.0, -1.0]) AS x |> SELECT SQRT(x) AS r
## safe-sqrt-ln-pow
FROM UNNEST([4.0, -1.0, 0.0]) AS x |> SELECT x, SAFE.SQRT(x) AS r, SAFE.LN(x) AS l, SAFE.POW(x, 0.5) AS p
## error-offset
FROM events |> SELECT tags[OFFSET(1)] AS second
## safe-offset
FROM events |> SELECT id, tags[SAFE_OFFSET(1)] AS second, tags[SAFE_ORDINAL(1)] AS first
## error-repeat
FROM sales |> SELECT REPEAT(product, units - 4) AS r
## error-substr
FROM sales |> SELECT SUBSTR(product, 1, units - 4) AS s
## window-in-where
FROM sales |> WHERE ROW_NUMBER() OVER (PARTITION BY region ORDER BY units DESC, product) = 1 |> SELECT region, product, units
## distinct-window
FROM sales |> EXTEND COUNT(DISTINCT product) OVER (PARTITION BY region) AS kinds |> SELECT DISTINCT region, kinds
## error-distinct-window-order
FROM sales |> EXTEND COUNT(DISTINCT product) OVER (ORDER BY month) AS k
## unnest-strict
FROM events |> CROSS JOIN UNNEST(tags AS a, tags AS b, mode => 'STRICT') |> SELECT id, a, b
## error-unnest-strict
FROM UNNEST([1, 2] AS a, [3] AS b, mode => 'STRICT')
## like-any-unnest
FROM sales |> WHERE product LIKE ANY UNNEST(['a%', 'p%r']) |> SELECT region, product
## not-like-all-unnest
FROM events |> SELECT id, kind NOT LIKE ALL UNNEST(['c%', '%w']) AS other, kind LIKE ALL UNNEST(tags) AS all_tags
## array-path-offset
FROM events, events.tags AS tag WITH OFFSET AS o |> SELECT id, tag, o
## left-join-array-path
FROM events LEFT JOIN events.tags AS tag |> SELECT id, tag
## left-join-unnest-on
FROM events LEFT JOIN UNNEST(events.tags) AS tag ON tag != 'a' |> SELECT id, tag
## having-max-rollup
FROM sales |> AGGREGATE ANY_VALUE(product HAVING MAX units) AS best, SUM(units HAVING MIN price) AS cheap, GROUPING(region) AS g GROUP BY ROLLUP(region)
## call-temp-table-function
CREATE TEMP TABLE FUNCTION top_n(src ANY TABLE, k INT64) AS (FROM src |> ORDER BY units DESC |> LIMIT k); FROM sales |> CALL top_n(2) |> SELECT product, units
## dot-star-expression
FROM sales |> SELECT STRUCT(region AS r, units AS u, price AS p).* EXCEPT (p) REPLACE (u * 2 AS u)
## dot-star-aggregate
FROM sales |> AGGREGATE ARRAY_AGG(STRUCT(product AS p, units AS u) ORDER BY units DESC LIMIT 1)[OFFSET(0)].*, COUNT(*) AS n GROUP BY region
## error-dot-star-not-aggregate
FROM UNNEST([STRUCT(1 AS a, 2 AS b)]) AS s |> AGGREGATE s.*
## select-as-struct
FROM sales |> SELECT AS STRUCT region, units |> WHERE units > 4 |> SELECT *
## select-as-struct-value
FROM sales |> WHERE units > 5 |> SELECT AS STRUCT region, units + 1
## array-of-structs
FROM UNNEST([1]) |> SELECT ARRAY(FROM sales |> WHERE region = 'west' |> SELECT AS STRUCT product, units |> ORDER BY units) AS arr
## select-as-value
FROM sales |> SELECT AS VALUE units |> SELECT *
## error-select-as-value
FROM sales |> SELECT AS VALUE units, price
## star-anonymous
FROM (FROM sales |> SELECT units + 1, region) |> SELECT *
## cast-format-dates
FROM sales |> SELECT DISTINCT month, CAST(month AS STRING FORMAT 'Mon DD, YYYY (Dy)') AS label, CAST(CAST(month AS STRING FORMAT 'YYYYMMDD') AS DATE FORMAT 'YYYYMMDD') AS back
## cast-format-timestamps
FROM events |> SELECT id, CAST(ts AS STRING FORMAT 'YYYY-MM-DD HH12:MI P.M. TZH:TZM' AT TIME ZONE 'Asia/Kolkata') AS text, UNIX_SECONDS(CAST(CAST(ts AS STRING FORMAT 'YYYY-MM-DD HH24:MI:SS' AT TIME ZONE 'UTC') AS TIMESTAMP FORMAT 'YYYY-MM-DD HH24:MI:SS' AT TIME ZONE 'UTC')) AS back
## cast-format-bytes
FROM sales |> SELECT DISTINCT product, CAST(CAST(product AS BYTES FORMAT 'UTF-8') AS STRING FORMAT 'BASE64') AS b64, CAST(CAST(product AS BYTES FORMAT 'ASCII') AS STRING FORMAT 'HEX') AS hex
## error-cast-format-parse
FROM sales |> SELECT CAST(product AS DATE FORMAT 'YYYY-MM-DD') AS d
## cast-format-safe
FROM sales |> SELECT DISTINCT product, SAFE_CAST(product AS DATE FORMAT 'YYYY-MM-DD') AS d
## cast-format-numbers
FROM sales |> SELECT product, CAST(units AS STRING FORMAT 'S0999') AS u, CAST(price AS STRING FORMAT '$9,999.99') AS p, CAST(price * 1000 AS STRING FORMAT 'FM9.99EEEE') AS e, CAST(units AS STRING FORMAT 'XX') AS hex
## error-cast-format-number
FROM sales |> SELECT CAST(units AS STRING FORMAT '9X') AS u
## intervals
FROM UNNEST([1]) |> SELECT INTERVAL 2 DAY AS d, INTERVAL '1-2' YEAR TO MONTH AS ym, DATE '2024-01-31' + INTERVAL 1 MONTH AS feb
## error-unknown-column
FROM sales |> WHERE nope = 1
## error-type
FROM sales |> WHERE region + 1 > 0
## error-aggregate-in-where
FROM sales |> WHERE SUM(units) > 3
## error-unknown-table
FROM nowhere
## error-mismatched-union
FROM sales |> SELECT region |> UNION ALL (FROM sales |> SELECT units)
## error-group-by-non-aggregate
FROM sales |> AGGREGATE units GROUP BY region
## error-function-signature
FROM sales |> SELECT SUBSTR(region)
